Using locomotion providers

To customize behavior of existing locomotion providers, VotanicXR2 allows developers to manually control locomotion providers under player entities which is accessed through VXRPlayer.Players.First().LocomotionWrapper.

Locomotion control API

The VXRLocomotionWrapper class provides the following methods for controlling locomotion providers of a player.

VXRLocomotionWrapper Public Methods

Method Description
GetLocomotionProviders Retrieves all locomotion providers of player.
GetActiveLocomotionProviders Retrieves all active locomotion providers of player.
GetLocomotionProvider Retrieves a locomotion provider of the specified type from the player. Returns null if a locomotion provider of that type is not found.
RegisterLocomotionProvider Add the specified locomotion provider to player's LocomotionMediator.
UnregisterLocomotionProvider Remove the specified locomotion provider from player's LocomotionMediator.

Code example: Press button to toggle locomotion providers

This example demonstrates how to implement toggling the locomotion providers when you press a button.

using System.Collections;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using UnityEngine.XR.Interaction.Toolkit.Locomotion;
using Votanic.VXR2.Core.Input;
using Votanic.VXR2.Core.Locomotion;
using Votanic.VXR2.Core;

public class SampleLocomotionController : MonoBehaviour
{
    private readonly string triggerActionName = "NextLocomotionProvider";

    private VXRActionBinding<bool> _buttonAction;
    private List<LocomotionProvider> locomotionProviders;
    private bool enableLocomotion = true;
    private VXRLocomotionWrapper locomotionWrapper;

    IEnumerator Start()
    {
        yield return new WaitUntil(() => VXRObjectManager.Instance.Initialized);
        _buttonAction = new VXRActionBinding<bool>(triggerActionName, ButtonAction);

        // Access locomotion wrapper of current player
        locomotionWrapper = VXRPlayer.Players.First().LocomotionWrapper;

        // Get all locomotion providers from locomotion wrapper
        locomotionProviders = locomotionWrapper.GetLocomotionProviders.ToList();
    }

    private void OnEnable()
    {
        if (_buttonAction != null) _buttonAction.Bound = true;
    }
    private void OnDisable()
    {
        if (_buttonAction != null) _buttonAction.Bound = false;
    }
    private void OnDestroy()
    {
        if (_buttonAction != null) _buttonAction.Dispose();
    }

    private void ButtonAction(bool value)
    {
        if (value) ToggleLocomotionProviders();
    }

    private void ToggleLocomotionProviders()
    {
        enableLocomotion = !enableLocomotion;
        
        if (enableLocomotion)
        {
            foreach (LocomotionProvider provider in locomotionProviders)
            {
                locomotionWrapper.RegisterLocomotionProvider(provider);
            }
        }
        else
        {
            foreach (LocomotionProvider provider in locomotionProviders)
            {
                locomotionWrapper.UnregisterLocomotionProvider(provider);
            }
        }
    }
}
